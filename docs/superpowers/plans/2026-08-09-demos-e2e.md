```ts
test("real API run is approved through the browser and completes", async ({ page, request }) => {
  const created = await request.post(`${apiBase}/api/runs`, { data: { scenario: "pending_write" } });
  expect(created.status()).toBe(201);
  const pending = await (await request.get(`${apiBase}/api/runs/${(await created.json()).id}`)).json();
  expect(pending.status).toBe("waiting_approval");
  expect(typeof pending.approval_id).toBe("string");
  await page.goto(uiBase);
  await expect(page.getByText("approval_pending")).toBeVisible();
  await page.getByRole("button", { name: "批准" }).click();
  await expect.poll(async () => (await (await request.get(`${apiBase}/api/runs/${pending.id}`)).json()).status).toBe("completed");
});

test("workbench remains usable at 320px", async ({ page, request }) => {
  await request.post(`${apiBase}/api/runs`, { data: { scenario: "pending_write" } });
  await page.setViewportSize({ width: 320, height: 720 });
  await page.goto(uiBase);
  await expect(page.getByRole("button", { name: "批准" })).toBeVisible();
  expect(await page.evaluate(() => document.documentElement.scrollWidth <= window.innerWidth)).toBeTruthy();
});
```
